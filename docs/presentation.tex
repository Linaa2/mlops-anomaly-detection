\begin{frame}{Context \& Problem}

\begin{columns}[T]
  \begin{column}{0.55\textwidth}
    \textbf{Use case:} Industrial predictive maintenance\\[6pt]
    \textbf{Dataset:} UCI --- Condition Monitoring of Hydraulic Systems
    \begin{itemize}
      \item 2205 operating cycles $\times$ 17 sensors
      \item 4 component conditions to predict simultaneously
      \item Labels from \texttt{profile.txt} (supervised)
    \end{itemize}
    \vspace{8pt}
    \textbf{Formulation:} Multi-class multi-output classification\\[4pt]
    {\small\color{tptgray}\faArrowRight\hspace{4pt} Why not anomaly detection? Labeled data available\\ $\to$ per-component diagnostics are more actionable}
  \end{column}

  \begin{column}{0.42\textwidth}
    \small
    \begin{tabular}{@{}lll@{}}
      \toprule
      \textbf{Target} & \textbf{Component} & \textbf{Classes} \\
      \midrule
      cooler & Cooler & 3 \\
      valve & Valve & 4 \\
      pump & Pump & 3 \\
      accumulator & Accum. & 4 \\
      \bottomrule
    \end{tabular}
    \vspace{10pt}

    \textbf{17 sensors:}\\
    {\footnotesize PS1--PS6, EPS1, FS1--FS2,\\TS1--TS4, VS1, CE, CP, SE}\\[4pt]
    {\footnotesize Feature eng.: mean per cycle}
  \end{column}
\end{columns}

\end{frame}

% SLIDE 4 — ML Pipeline & Continuous Training (1m30)
% Justify: model choice, CT on static data, promote/reject logic
% ═══════════════════════════════════════════════════════════════
\begin{frame}{ML Pipeline \& Continuous Training}

\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \textbf{Model:} \texttt{MultiOutputClassifier(RF)}\\
    {\scriptsize\color{tptgray} Why? Single training run, consistent features, simpler deployment than 4 models}\\[4pt]

    \textbf{Results (test set 20\%):}\\[2pt]
    \small
    \begin{tabular}{@{}lccc@{}}
      \toprule
      \textbf{Target} & \textbf{Acc.} & \textbf{F1\textsubscript{macro}} & \textbf{F1\textsubscript{w.}} \\
      \midrule
      Cooler     & 1.000 & 1.000 & 1.000 \\
      Valve      & 0.948 & 0.947 & 0.948 \\
      Pump       & 0.993 & 0.993 & 0.993 \\
      Accumulator & 0.990 & 0.990 & 0.990 \\
      \bottomrule
    \end{tabular}

    \vspace{6pt}
    \textbf{Design decisions:}
    \begin{itemize}\scriptsize\setlength\itemsep{1pt}
      \item F1 macro for comparison (handles class imbalance)
      \item DAG delegates to \texttt{src/train.py} (separation of concerns)
      \item Promote only if F1\textsubscript{new} $>$ F1\textsubscript{prod}
    \end{itemize}
  \end{column}

  \begin{column}{0.45\textwidth}
    \textbf{CT Logic (2 DAGs):}\\[4pt]
    \footnotesize
    \begin{tikzpicture}[
      node distance=0.4cm,
      task/.style={draw, rounded corners=3pt, minimum width=2.8cm, minimum height=0.5cm, font=\scriptsize, fill=tptblue!8, draw=tptblue!50},
      arr/.style={-{Stealth[length=4pt]}, tptblue!60},
    ]
      \node[task] (dl) {download\_dataset};
      \node[task, below=of dl] (uz) {unzip};
      \node[task, below=of uz] (mg) {merge\_sensors};
      \node[task, below=of mg] (pp) {preprocess};
      \node[task, below=of pp] (sa) {sample 80\%};
      \node[task, below=of sa] (tr) {trigger\_training};

      \draw[arr] (dl)--(uz); \draw[arr] (uz)--(mg);
      \draw[arr] (mg)--(pp); \draw[arr] (pp)--(sa);
      \draw[arr] (sa)--(tr);

      \node[task, right=0.6cm of sa, fill=warnor!10, draw=warnor!50] (tm) {train\_model};
      \node[task, below=of tm, fill=warnor!10, draw=warnor!50] (pr) {promote\_or\_reject};

      \draw[arr, warnor!60] (tr.east) -- ++(0.2,0) |- (tm.west);
      \draw[arr, warnor!60] (tm)--(pr);

      \node[font=\scriptsize\color{tptgray}, above=0.1cm of dl] {@daily};
      \node[font=\scriptsize\color{tptgray}, above=0.1cm of tm] {event-driven};
    \end{tikzpicture}

    \vspace{6pt}
    {\scriptsize \textbf{Why 80\% sampling on static data?}\\
    Simulates data variability $\to$ champion/challenger\\comparison becomes meaningful.}
  \end{column}
\end{columns}

\end{frame}

      \bottomrule
    \end{tabular}

    \vspace{6pt}
    \textbf{Git workflow:}
    \begin{itemize}\small
      \item Feature branches $\to$ PR $\to$ main
      \item CI gates on every PR (ruff + pytest)
      \item 20+ PRs, conventional commits
      \item Copilot code reviews
    \end{itemize}
  \end{column}

  \begin{column}{0.48\textwidth}
    \textbf{Honest limitations:}
    \begin{itemize}\small\setlength\itemsep{3pt}
      \item Static dataset --- CT designed for production but validated on sampled data
      \item No K8s cluster --- CD ready, deploy skipped
      \item API loads \texttt{model.pkl} locally\\{\scriptsize (prod: would query MLflow Registry)}
      \item Heavy DAG imports $\to$ deferred inside task functions (30s parsing limit)
    \end{itemize}

    \vspace{4pt}
    \textbf{Key design principle:}\\
    {\small Separation of concerns:\\
    \texttt{src/} = business logic, \texttt{dags/} = orchestration,\\
    \texttt{api/} = serving, \texttt{webapp/} = UI}
  \end{column}
\end{columns}
