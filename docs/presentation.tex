    \begin{itemize}

    {\small\color{tptgray}\faArrowRight\hspace{4pt} Pourquoi pas d\'etection d'anomalies ? Donn\'ees labellis\'ees\\ $\to$ diagnostic par composant plus actionnable}
  \end{column}

  \begin{column}{0.42\textwidth}
    \small
    \begin{tabular}{@{}lll@{}}
      \toprule
      \textbf{Cible} & \textbf{Composant} & \textbf{Classes} \\
      \midrule
      cooler & Refroidisseur & 3 \\
      valve & Vanne & 4 \\
      pump & Pompe & 3 \\
      accumulator & Accumulateur & 4 \\
      \bottomrule
    \end{tabular}
    \vspace{10pt}

    \textbf{17 capteurs :}\\
    {\footnotesize PS1--PS6, EPS1, FS1--FS2,\\TS1--TS4, VS1, CE, CP, SE}\\[4pt]
    {\footnotesize Feature eng. : moyenne par cycle}
  \end{column}
\end{columns}

\end{frame}

\begin{frame}{Pipeline ML \& Continuous Training}

\begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \textbf{Mod\`ele :} \texttt{MultiOutputClassifier(RF)}\\
    {\scriptsize\color{tptgray} Un seul entra\^inement, d\'eploiement plus simple que 4 mod\`eles}\\[3pt]

    \textbf{MLflow --- Experiment Tracking :}\\[2pt]
    \centering
    \includegraphics[width=\linewidth, height=3.3cm, keepaspectratio]{mlflow.jpeg}\\[-1pt]
    {\tiny Run MLflow --- 4 m\'etriques F1 macro, source \texttt{train.py}}

    \vspace{3pt}
    \raggedright
    \textbf{Choix de conception :}
    \begin{itemize}\scriptsize\setlength\itemsep{1pt}
      \item F1 macro pour la comparaison (g\`ere le d\'es\'equilibre)
      \item DAG d\'el\`egue \`a \texttt{src/train.py} (s\'eparation des resp.)
      \item Promotion seulement si F1\textsubscript{new} $>$ F1\textsubscript{prod}
    \end{itemize}
  \end{column}

  \begin{column}{0.45\textwidth}
    \textbf{Logique CT (2 DAGs) :}\\[2pt]
    \footnotesize
    \begin{tikzpicture}[
      node distance=0.2cm,

      \node[task, below=of pp] (sa) {sample 80\%};
      \node[task, below=of sa] (tr) {trigger\_training};

      \draw[arr] (dl)--(uz); \draw[arr] (uz)--(mg);
      \draw[arr] (mg)--(pp); \draw[arr] (pp)--(sa);
      \draw[arr] (sa)--(tr);

      \node[task, right=0.5cm of sa, fill=warnor!10, draw=warnor!50] (tm) {train\_model};
      \node[task, below=of tm, fill=warnor!10, draw=warnor!50] (pr) {promote\_or\_reject};

      \draw[arr, warnor!60] (tr.east) -- ++(0.15,0) |- (tm.west);
      \draw[arr, warnor!60] (tm)--(pr);

      \node[font=\tiny\color{tptgray}, above=0.05cm of dl] {@daily};
      \node[font=\tiny\color{tptgray}, above=0.05cm of tm] {event-driven};
    \end{tikzpicture}

    \vspace{4pt}
    {\scriptsize \textbf{Pourquoi \'echantillonner 80\% ?}\\
    Dataset statique $\to$ l'\'echantillonnage simule\\la variabilit\'e, rendant la comparaison\\champion/challenger pertinente.}
  \end{column}
\end{columns}
